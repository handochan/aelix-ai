def _channel(scene, root: Path) -> PrintChannel:
    home = root / "child-home"
    (home / ".config").mkdir(parents=True, exist_ok=True)
    agent_dir = home / "agent"
    agent_dir.mkdir(parents=True, exist_ok=True)
    return PrintChannel(env_builder=lambda profile: build_child_env(
        profile,
        base=child_env(home, XDG_CONFIG_HOME=str(home / ".config"),
                      AELIX_CODING_AGENT_DIR=str(agent_dir),
                      REVIEW_KEY="dummy-no-network",
                      PI_OFFLINE="1", PYTHONDONTWRITEBYTECODE="1"),
    ))


async def _hook_only(scene, task="keep going", key="tc-resume"):
    args = _resume_args(scene, task)
    result = await scene.bench.hook(ToolCallHookEvent(
        tool_call_id=key, tool_name="agent", args=args,
    ))
    assert result is None, result
    return args


async def _execute(scene, args, key="tc-resume"):
    return await scene.bench.tool.execute(args, ToolExecutionContext(tool_call_id=key))

async def test_hard_link_alias_evades_writer_lock_and_writes_outside_bucket(tmp_path):

    scene, capture, settings = await _real_scene(tmp_path, model="m1")

    assert other_writer.try_acquire()
    try:

        target = await scene.bench.ext.runtime.find_resume_target(scene.child_id)
        result = await asyncio.wait_for(_resume(scene), 20)

        assert not result.is_error, _text(result)
        assert outside.stat().st_size > before
    finally:
        other_writer.release()

@pytest.mark.parametrize("adapter", [
    "anthropic-messages", "openai-completions", "openai-responses",
    "openai-codex-responses", "google-generative-ai", "google-vertex",
])
def test_builtin_wire_converters_keep_dangling_call_as_error(adapter):
    from aelix_ai.messages import AssistantMessage, UserMessage, TextContent, ToolCallContent
    from aelix_ai.streaming import Context, Model
    from aelix_ai.providers._transform_messages import transform_messages
    from aelix_ai.providers._anthropic_transforms import transform_messages as anthropic
    from aelix_ai.providers._google_shared import convert_messages as google
    from aelix_ai.providers._openai_responses_shared import convert_responses_messages
    from aelix_ai.providers._openai_compat import get_compat
    from aelix_ai.providers.openai_completions import convert_messages as completions
    model = Model(id="m", provider="review", api=adapter, input=["text"])
    messages = [UserMessage(content=[TextContent(text="first task")]),
        AssistantMessage(content=[ToolCallContent(tool_call_id="call_1", tool_name="bash", input={"command": "sleep 3"})],
            stop_reason="tool_use", provider=model.provider, model=model.id, api=adapter),
        UserMessage(content=[TextContent(text="Task: continue")])]
    normalized = transform_messages(messages, model)
    synthetic = [m for m in normalized if m.role == "toolResult"]
    assert len(synthetic) == 1 and synthetic[0].is_error
    context = Context(messages=messages)
    if adapter == "anthropic-messages":
        wire = anthropic(messages, model)
        results = [b for m in wire for b in m.get("content", []) if isinstance(b, dict) and b.get("type") == "tool_result"]
        assert results[0]["is_error"] is True
    elif adapter in ("google-generative-ai", "google-vertex"):
        wire = google(model, context)
        results = [b["functionResponse"] for m in wire for b in m.get("parts", []) if "functionResponse" in b]
        assert "error" in results[0]["response"]
    elif adapter in ("openai-responses", "openai-codex-responses"):
        wire = convert_responses_messages(model, context)
        results = [m for m in wire if m.get("type") == "function_call_output"]
        assert len(results) == 1 and "No result provided" in str(results[0])
    else:
        wire = completions(model, context, get_compat(model))
        results = [m for m in wire if m.get("role") == "tool"]
        assert len(results) == 1 and "No result provided" in str(results[0])
    print(f"ORPHAN adapter={adapter} synthetic_error={synthetic[0].is_error} wire_result={results}")


async def test_fresh_argv_matches_base_matrix(tmp_path):

        target.write_text(source)
        spec = importlib.util.spec_from_file_location(name, target)
        module = importlib.util.module_from_spec(spec)
        sys.modules[name] = module
        spec.loader.exec_module(module)

    channel = old_module("aelix_agents.review_610_old_channel", "packages/aelix-coding-agent/src/aelix_agents/print_channel.py")

    r = await registry(tmp_path)
    r.register_provider("own", ProviderConfigInput(models={"m1": row("own")}))
    compared = 0
    parent_cwd = str(tmp_path.resolve())
    # Real emitter functions from f3e76fbb versus the new fresh-limits path.
    for tools, parent_tools, contexts, thinking, identity, mode, source in itertools.product(
        [None, (), ("read", "write", "bash")],
        [None, (), ("read", "write")],
        [(True, True), (True, False), (False, True)],
        [(None, None), (None, "high"), ("low", "high")],
        [(None, None), ("m1", "own"), ("own/m1", None)],
        [PermissionMode.PLAN, PermissionMode.YOLO],
        ["append", "replace"],

            channel.narrow_tools(profile, parent_tools).profile, contexts[1]), thinking[1])
        common = dict(prompt_path=str(tmp_path / "prompt.md"), task="--literal task", permission_mode=mode,

            parent_context_files=contexts[1], parent_thinking=thinking[1], parent_model=parent_model, model_registry=r).limits
        after = build_child_argv(limited_profile(profile, limits), **common, model_pin=limits.model)
        assert before == after, (before, after)
        compared += 1
    print(f"FRESH_ARGV compared={compared} byte_identical=True")

def test_preflight_huge_file_read_is_bounded(tmp_path):
    from aelix_agents.child_session import CHILD_SCAN_BYTES, _read_head
    large = tmp_path / "huge.jsonl"
    with large.open("wb") as output:
        output.write(b"prefix\n")
        output.truncate(32 * 1024 * 1024)
    head, complete = _read_head(str(large))
    print(f"HUGE file_bytes={large.stat().st_size} scanned_bytes={len(head)} complete={complete}")
    assert len(head) == CHILD_SCAN_BYTES and not complete
